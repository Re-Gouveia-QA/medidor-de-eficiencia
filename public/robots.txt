# Só as páginas públicas de auth valem indexação (o resto do app exige login e não tem
# conteúdo indexável mesmo sem esta regra — isto é defesa em profundidade, não a única barreira).
# Bloqueio geral primeiro, allowlist específica depois: uma rota privada nova adicionada no
# futuro fica escondida por padrão, em vez de precisar lembrar de listá-la aqui.
User-agent: *
Disallow: /
# "/" (a home pública, ver plano de landing page) precisa do "$" de fim-de-string (extensão
# aceita por Google/Bing) pra casar só a raiz exata — sem ele, "Allow: /" combinaria com QUALQUER
# caminho como prefixo (o mesmo que "Disallow: /" já combina), empatando em todas as rotas e
# efetivamente anulando o bloqueio geral acima, não só liberando a home.
Allow: /$
Allow: /login
Allow: /register
Allow: /forgot-password
# Contém token no path — nunca deve ser crawleada/indexada, mesmo sendo tecnicamente pública
# antes da validação do token (Disallow explícito, não coberto pelo "Disallow: /" acima porque
# uma Allow mais específica de /login etc. não vaza para cá, mas fica documentado por clareza).
Disallow: /reset-password/
